import { Card, CardGrid, LinkCard } from "@astrojs/starlight/components";

<CardGrid>
  <Card title="Self-hosted in three steps" icon="rocket">
    Copy `.env.example` to `.env`, set `PUBLIC_URL`, run `docker compose up -d`. One app container and PostgreSQL 17. No Redis, no queue, no cloud account.
  </Card>
  <Card title="Singapore payroll" icon="document">
    Monthly runs with CPF (OW and AW ceilings, SPR rates), SHG funds, SDL and an FWL estimate; itemised payslips; the CPF EZPay file, a bank file and an IR8A export to upload yourself.
  </Card>
  <Card title="Shown working" icon="magnifier">
    Every CPF figure shows its working, the rate table it used and a link to the source. The statutory engine is tested against CPF's published examples and its online calculator.
  </Card>
  <Card title="Your sign-in, your rules" icon="padlock">
    Email and password with two-step verification, Google Workspace, Microsoft Entra ID, any OpenID Connect provider, and Singpass (needs your own Singpass onboarding).
  </Card>
</CardGrid>
